cmake_minimum_required(VERSION 3.22)
project(zerolistpool)
set(ZL "${CMAKE_CURRENT_SOURCE_DIR}/../../../../")
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(ZIG_DIR "android-arm64")
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(ZIG_DIR "android-x86_64")
else()
  message(FATAL_ERROR "ZeroList: unsupported ABI ${ANDROID_ABI}")
endif()
add_library(pool_zig_engine STATIC IMPORTED)
set_target_properties(pool_zig_engine PROPERTIES IMPORTED_LOCATION "${ZL}/zig/zig-out/${ZIG_DIR}/libzerolist_engine.a")
add_library(zerolistpool SHARED poolbridge.cpp)
target_include_directories(zerolistpool PRIVATE "${ZL}/cpp")
target_link_libraries(zerolistpool pool_zig_engine log)
